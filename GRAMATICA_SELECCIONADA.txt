file:
    | statements tk_eof

statements:
    | statement statements_tail
    | COMMENT NEWLINE statements_tail
    | NEWLINE statements_tail
    | ε

statements_tail:
    | statement statements_tail
    | COMMENT NEWLINE statements_tail
    | NEWLINE statements_tail
    | ε

statement:
    | simple_stmt
    | compound_stmt

simple_stmt:
    | small_stmt simple_stmt_tail

small_stmt:
    | assignment_stmt
    | return_stmt
    | import_stmt
    | pass_stmt
    | break_stmt
    | continue_stmt
    | del_stmt
    | assert_stmt
    | global_stmt
    | nonlocal_stmt
    | raise_stmt

simple_stmt_tail:
    | tk_punto_y_coma small_stmt simple_stmt_tail
    | ε

compound_stmt:
    | function_def
    | class_def
    | if_stmt
    | while_stmt
    | for_stmt
    | try_stmt
    | match_stmt

assignment_stmt:
    | target_assignment assignment_suffix

target_assignment:
    | NAME
    | tk_par_izq single_target tk_par_der
    | single_subscript_attribute_target

assignment_suffix:
    | tk_dos_puntos expression annotated_assignment_opt
    | tk_asig annotated_rhs assignment_chain_opt 
    | augassign annotated_rhs
    | ε

annotated_assignment_opt:
    | tk_asig annotated_rhs
    | ε

annotated_rhs:
    | expression

assignment_chain_opt:
    | tk_asig target_assignment assignment_chain_opt
    | ε

augassign:
    | tk_mas_asig | tk_menos_asig | tk_mult_asig | tk_arroba_asig
    | tk_div_asig | tk_mod_asig | tk_amper_asig | tk_bar_asig
    | tk_hat_asig | tk_left_shift_asig | tk_right_shift_asig
    | tk_pot_asig | tk_div_entera_asig

return_stmt:
    | 'return' return_expr_opt

return_expr_opt:
    | expression
    | ε

pass_stmt: | 'pass'
break_stmt: | 'break'
continue_stmt: | 'continue'

del_stmt:
    | 'del' del_targets del_stmt_terminator

del_stmt_terminator:
    | tk_punto_y_coma
    | NEWLINE

del_targets:
    | del_target del_targets_tail

del_targets_tail:
    | tk_coma del_target del_targets_tail
    | tk_coma
    | ε

del_target:
    | atom noncall_trailer_seq
    | tk_par_izq del_target tk_par_der
    | tk_par_izq del_targets tk_par_der
    | tk_corchete_izq del_targets tk_corchete_der

assert_stmt:
    | 'assert' expression assert_stmt_expression
assert_stmt_expression:
    | tk_coma expression
    | ε

global_stmt:
    | 'global' NAME global_name_list_tail
global_name_list_tail:
    | tk_coma NAME global_name_list_tail
    | ε

nonlocal_stmt:
    | 'nonlocal' NAME nonlocal_name_list_tail
nonlocal_name_list_tail:
    | tk_coma NAME nonlocal_name_list_tail
    | ε

raise_stmt:
    | 'raise' raise_stmt_tail
raise_stmt_tail:
    | expression from_expression
    | ε
from_expression:
    | 'from' expression
    | ε

function_def:
    | 'def' NAME tk_par_izq params tk_par_der function_return tk_dos_puntos block
    | 'async' 'def' NAME tk_par_izq params tk_par_der function_return tk_dos_puntos block

function_return:
    | tk_ejecuta expression
    | ε

params:
    | param param_list_tail
    | ε

param:
    | NAME param_tail
param_tail:
    | tk_dos_puntos expression param_default_opt
    | tk_asig expression
    | ε
param_default_opt:
    | tk_asig expression
    | ε

param_list_tail:
    | tk_coma param param_list_tail
    | tk_coma
    | ε
class_def:
    | 'class' NAME class_args_opt tk_dos_puntos block

class_args_opt:
    | tk_par_izq arguments tk_par_der
    | ε
if_stmt:
    | 'if' expression tk_dos_puntos block elif_else_part

elif_else_part:
    | 'elif' expression tk_dos_puntos block elif_else_part
    | 'else' tk_dos_puntos block
    | ε
while_stmt:
    | 'while' expression tk_dos_puntos block else_block

for_stmt:
    | for_prefix targets 'in' expression tk_dos_puntos block else_block

for_prefix:
    | 'for'
    | 'async' 'for'

else_block:
    | 'else' tk_dos_puntos block
    | ε
try_stmt:
    | 'try' tk_dos_puntos block try_continuation

try_continuation:
    | finally_block
    | except_block_list else_block finally_block

except_block_list:
    | except_block except_block_list_tail

except_block_list_tail:
    | except_block except_block_list_tail
    | ε

except_block:
    | 'except' except_header tk_dos_puntos block

except_header:
    | expression 'as' NAME
    | expression
    | expressions
    | ε

finally_block:
    | 'finally' tk_dos_puntos block
    | ε

match_stmt:
    | 'match' subject_expr tk_dos_puntos NEWLINE INDENT case_block_list DEDENT

subject_expr:
    | named_expression subject_expr_tail
subject_expr_tail:
    | tk_coma named_expression subject_expr_tail_opt_comma
    | ε
subject_expr_tail_opt_comma:
    | tk_coma
    | ε

case_block_list:
    | case_block case_block_list_tail
case_block_list_tail:
    | case_block case_block_list_tail
    | ε

case_block:
    | 'case' patterns guard_opt tk_dos_puntos block

guard_opt:
    | guard
    | ε

guard:
    | 'if' named_expression

patterns:
    | pattern

pattern:
    | or_pattern pattern_as_opt

pattern_as_opt:
    | 'as' pattern_capture_target
    | ε

or_pattern:
    | closed_pattern or_pattern_tail
or_pattern_tail:
    | tk_or closed_pattern or_pattern_tail
    | ε

closed_pattern:
    | literal_pattern
    | capture_pattern
    | value_pattern
    | group_pattern
    | sequence_pattern
    | mapping_pattern
    | class_pattern

literal_pattern:
    | signed_number
    | string
    | 'None'
    | 'True'
    | 'False'

capture_pattern:
    | pattern_capture_target

pattern_capture_target:
    | NAME

value_pattern:
    | attr

attr:
    | name_or_attr tk_punto NAME

name_or_attr:
    | attr
    | NAME

group_pattern:
    | tk_par_izq pattern tk_par_der

sequence_pattern:
    | tk_corchete_izq pattern_list_opt tk_corchete_der
    | tk_par_izq pattern_list_opt tk_par_der

pattern_list_opt:
    | pattern pattern_list_rest
    | ε

pattern_list_rest:
    | tk_coma pattern pattern_list_rest
    | tk_coma
    | ε

mapping_pattern:
    | tk_llave_izq mapping_items tk_llave_der
mapping_items:
    | ε
    | key_value_list
    | key_value_list tk_coma

key_value_list:
    | key_value_pair key_value_list_tail
key_value_list_tail:
    | tk_coma key_value_pair key_value_list_tail
    | ε

key_value_pair:
    | literal_expr tk_dos_puntos pattern
    | attr tk_dos_puntos pattern

class_pattern:
    | name_or_attr tk_par_izq class_pattern_body tk_par_der

class_pattern_body:
    | ε
    | positional_patterns class_pattern_body_rest
    | keyword_patterns class_pattern_body_comma_opt

class_pattern_body_rest:
    | tk_coma keyword_patterns class_pattern_body_comma_opt
    | class_pattern_body_comma_opt

class_pattern_body_comma_opt:
    | tk_coma
    | ε

positional_patterns:
    | pattern positional_patterns_tail
positional_patterns_tail:
    | tk_coma pattern positional_patterns_tail
    | ε

keyword_patterns:
    | keyword_pattern keyword_patterns_tail
keyword_patterns_tail:
    | tk_coma keyword_pattern keyword_patterns_tail
    | ε

keyword_pattern:
    | NAME tk_asig pattern

expressions:
    | expression expressions_tail
expressions_tail:
    | tk_coma expression expressions_tail
    | tk_coma
    | ε

expression:
    | disjunction 'if' disjunction 'else' expression
    | disjunction

assignment_expression:
    | NAME tk_colon_asig expression

named_expression:
    | assignment_expression
    | expression

disjunction:
    | conjunction disjunction_tail
disjunction_tail:
    | 'or' conjunction disjunction_tail
    | ε

conjunction:
    | inversion conjunction_tail
conjunction_tail:
    | 'and' inversion conjunction_tail
    | ε

inversion:
    | 'not' inversion
    | comparison

comparison:
    | bitwise_or comparison_tail

comparison_tail:
    | comp_op bitwise_or comparison_tail
    | ε

comp_op:
    | tk_igual | tk_distinto | tk_menor | tk_menor_igual | tk_mayor | tk_mayor_igual
    | 'in' | 'not' 'in' | 'is' | 'is' 'not'

bitwise_or:
    | bitwise_xor bitwise_or_tail
bitwise_or_tail:
    | tk_or bitwise_xor bitwise_or_tail
    | ε

bitwise_xor:
    | bitwise_and bitwise_xor_tail
bitwise_xor_tail:
    | tk_xor bitwise_and bitwise_xor_tail
    | ε

bitwise_and:
    | shift_expr bitwise_and_tail
bitwise_and_tail:
    | tk_and shift_expr bitwise_and_tail
    | ε

shift_expr:
    | sum shift_expr_tail
shift_expr_tail:
    | tk_left_shift sum shift_expr_tail
    | tk_right_shift sum shift_expr_tail
    | ε

sum:
    | term sum_tail
sum_tail:
    | tk_suma term sum_tail
    | tk_resta term sum_tail
    | ε

term:
    | factor term_tail
term_tail:
    | tk_mult factor term_tail
    | tk_div factor term_tail
    | tk_div_entera factor term_tail
    | tk_modulo factor term_tail
    | tk_arroba factor term_tail
    | ε

factor:
    | tk_suma factor
    | tk_resta factor
    | tk_tilde factor
    | power

power:
    | await_primary power_tail
power_tail:
    | tk_potencia factor
    | ε

await_primary:
    | 'await' primary
    | atom trailer_seq

primary:
    | atom trailer_seq
trailer_seq:
    | trailer trailer_seq
    | ε
trailer:
    | tk_punto NAME
    | tk_par_izq arguments_opt tk_par_der
    | tk_corchete_izq slices tk_corchete_der

arguments_opt:
    | arguments
    | ε

arguments:
    | arg_list arguments_comma
    | ε
arguments_comma:
    | tk_coma
    | ε
arg_list:
    | argument_item arg_list_tail
    | ε
argument_item:
    | NAME tk_asig expression
    | expression
arg_list_tail:
    | tk_coma argument_item arg_list_tail
    | ε

atom:
    | NAME
    | tk_True
    | tk_False
    | tk_None
    | STRING
    | NUMBER
    | tk_par_izq expressions tk_par_der
    | tk_corchete_izq expressions tk_corchete_der
    | tk_llave_izq collection_content tk_llave_der
    | tk_ellipsis

collection_content:
    | kvpair kvpair_list_tail
    | expression set_list_tail
    | ε

kvpair_list_tail:
    | tk_coma kvpair kvpair_list_tail
    | ε

kvpair:
    | expression tk_dos_puntos expression

set_list_tail:
    | tk_coma expression set_list_tail
    | ε

slices:
    | slice slices_tail
slices_tail:
    | tk_coma slice slices_tail
    | ε
slice:
    | slice_index
    | named_expression
slice_index:
    | expression_opt tk_dos_puntos expression_opt slice_step_opt
expression_opt:
    | expression
    | ε
slice_step_opt:
    | tk_dos_puntos expression_opt
    | ε
block:
    | NEWLINE INDENT statements DEDENT
