# Raíz del programa
program:
    | stmts_top ENDMARKER

# Sentencias en el nivel superior
stmts_top:
    | stmt stmts_top_prime

stmts_top_prime:
    | stmt stmts_top_prime
    | ε

# Sentencia general
stmt:
    | simple_stmts
    | complx

# Sentencias simples
simple_stmts:
    | simple_stmt NEWLINE

simple_stmt:
    | id assign_stmt
    | self assign_stmt
    | return_stmt
    | print_stmt
    | import_stmt
    | pass

# Sentencias complejas
complx:
    | def def_stmt
    | if_stmt
    | while_stmt
    | for_stmt
    | class_stmt
    | try_stmt

# Definiciones de funciones
def_stmt:
    | def_init_stmt
    | def_normal_stmt

def_init_stmt:
    | __init__ tk_par_izq param_list tk_par_der tk_dos_puntos block

def_normal_stmt:
    | id tk_par_izq param_list tk_par_der tk_dos_puntos block

param_list:
    | id param_structure param_list_prime
    | self param_structure param_list_prime
    | ε

param_list_prime:
    | tk_coma param_list
    | ε

param_structure:
    | tk_dos_puntos param_list_complex
    | ε

param_list_complex:
    | tipo_lista tk_corchete_izq lista_tipos_datos tk_corchete_der
    | tk_corchete_izq tipos_datos tk_corchete_der
    | tipos_datos
    | ε

lista_tipos_datos:
    | tipos_datos lista_tipos_datos_prime

lista_tipos_datos_prime:
    | tk_coma tipos_datos lista_tipos_datos_prime
    | ε

tipos_datos:
    | int
    | float
    | str
    | bool

tipo_lista:
    | list
    | set
    | dict
    | tuple

# Bloques indentados
block:
    | NEWLINE TAB stmts TABend
    | simple_stmts

# Sentencias dentro de bloques
stmts:
    | stmt stmts_prime

stmts_prime:
    | stmt stmts_prime
    | ε

# Retorno, print, import
return_stmt:
    | return expr

print_stmt:
    | print tk_par_izq print_expr tk_par_der

print_expr:
    | expr print_tail

print_tail:
    | tk_coma expr print_tail
    | ε

import_stmt:
    | import id_list
    | from id import id_list

# Clases
class_stmt:
    | class id class_body

class_body:
    | tk_dos_puntos block
    | tk_par_izq id tk_par_der tk_dos_puntos block

# Identificadores y asignaciones
id_list:
    | id id_list_prime

id_list_prime:
    | tk_coma id id_list_prime
    | as id
    | ε

assign_stmt:
    | assign_op expr
    | factor_tail_prime assignment

assignment:
    | assign_op expr
    | ε

assign_op:
    | tk_asig
    | tk_mas_asig
    | tk_menos_asig
    | tk_mult_asig
    | tk_div_asig
    | tk_mod_asig

# Condiciones y bucles
condition:
    | expr condition_prime
    | tk_par_izq condition tk_par_der
    | True
    | False

condition_prime:
    | in id
    | comp_op expr
    | ε

comp_op:
    | tk_igual
    | tk_distinto
    | tk_menor
    | tk_mayor
    | tk_menor_igual
    | tk_mayor_igual

# Expresiones aritméticas y lógicas
expr:
    | term expr_prime

expr_prime:
    | tk_suma term expr_prime
    | tk_resta term expr_prime
    | tk_modulo term expr_prime
    | and term expr_prime
    | or term expr_prime
    | ε

term:
    | factor term_prime

term_prime:
    | ε
    | tk_mult factor term_prime
    | tk_div factor term_prime
    | tk_modulo factor term_prime

factor:
    | tk_par_izq expr tk_par_der
    | id factor_tail_prime
    | self factor_tail_prime
    | set_function
    | list_function
    | tuple_function
    | tk_corchete_izq items_array tk_corchete_der
    | tk_llave_izq set_dict tk_llave_der
    | num
    | True
    | False
    | not factor
    | tk_par_izq items_tuple tk_par_der
    | tk_cadena

factor_tail_prime:
    | tk_corchete_izq items_array tk_corchete_der factor_tail_prime
    | tk_par_izq arg_list tk_par_der factor_tail_prime
    | tk_punto id factor_tail_prime
    | ε

# Funciones de colección
set_dict:
    | items_set
    | items_dict
    | ε

items_set:
    | items_prime

items_dict:
    | pair dict_prime

pair:
    | expr tk_dos_puntos expr

dict_prime:
    | tk_coma pair dict_prime
    | ε

items_tuple:
    | expr items_prime

items_prime:
    | tk_coma items_rest_tail items_prime
    | ε

items_rest_tail:
    | expr
    | ε

items_array:
    | expr items_array_prime
    | ε

items_array_prime:
    | tk_coma items_array
    | tk_dos_puntos items_array_tail
    | ε

items_array_tail:
    | expr items_array_prime
    | ε

set_function:
    | set tk_par_izq items tk_par_der

list_function:
    | list tk_par_izq items tk_par_der

tuple_function:
    | tuple tk_par_izq items tk_par_der

items:
    | tk_cadena
    | tk_corchete_izq items_array tk_corchete_der factor_tail_prime
    | tk_par_izq arg_list tk_par_der factor_tail_prime
    | ε

arg_list:
    | expr arg_list_prime

arg_list_prime:
    | tk_coma expr arg_list_prime
    | ε

# Números
num:
    | tk_entero
    | tk_entero tk_punto tk_entero
    | tk_punto tk_entero
