programa:
    | NEWLINE programa
    | sentencias ENDMARKER

# ===================================================================
# SENTENCIAS
# ===================================================================

sentencias:
    | sentencia sentencias
    | ε

sentencia:
    | sentencia_simple NEWLINE
    | sentencia_compuesta

sentencia_simple:
    | asignacion
    | expresion
    | return_stmt
    | break
    | continue
    | pass
    | del_stmt
    | import_stmt
    | print_stmt

# ===================================================================
# ASIGNACIÓN
# ===================================================================

asignacion:
    | id op_asignacion expresion
    | id tk_corchete_izq expresion tk_corchete_der tk_asig expresion

op_asignacion:
    | tk_asig
    | tk_mas_asig
    | tk_menos_asig
    | tk_mult_asig
    | tk_div_asig
    | tk_mod_asig

# ===================================================================
# OTRAS SENTENCIAS SIMPLES
# ===================================================================

return_stmt:
    | return expresion_opt

expresion_opt:
    | expresion
    | ε

del_stmt:
    | del id

import_stmt:
    | import id import_as
    | from id import id import_as

import_as:
    | as id
    | ε

print_stmt:
    | print tk_par_izq args_opt tk_par_der

# ===================================================================
# SENTENCIAS COMPUESTAS
# ===================================================================

sentencia_compuesta:
    | if_stmt
    | while_stmt
    | for_stmt
    | def_stmt
    | class_stmt
    | try_stmt

# ===================================================================
# IF-ELIF-ELSE
# ===================================================================

if_stmt:
    | if expresion tk_dos_puntos bloque elif_chain else_opt

elif_chain:
    | elif expresion tk_dos_puntos bloque elif_chain
    | ε

else_opt:
    | else tk_dos_puntos bloque
    | ε

# ===================================================================
# WHILE
# ===================================================================

while_stmt:
    | while expresion tk_dos_puntos bloque

# ===================================================================
# FOR
# ===================================================================

for_stmt:
    | for id in expresion tk_dos_puntos bloque

# ===================================================================
# DEFINICIÓN DE FUNCIONES
# ===================================================================

def_stmt:
    | def id tk_par_izq parametros tk_par_der tk_dos_puntos bloque

parametros:
    | parametro lista_parametros
    | ε

lista_parametros:
    | tk_coma parametro lista_parametros
    | ε

parametro:
    | id param_default

param_default:
    | tk_asig expresion
    | ε

# ===================================================================
# DEFINICIÓN DE CLASES
# ===================================================================

class_stmt:
    | class id herencia_opt tk_dos_puntos bloque

herencia_opt:
    | tk_par_izq id tk_par_der
    | ε

# ===================================================================
# TRY-EXCEPT-FINALLY
# ===================================================================

try_stmt:
    | try tk_dos_puntos bloque except_clauses finally_opt

except_clauses:
    | except except_tipo tk_dos_puntos bloque except_clauses
    | ε

except_tipo:
    | id
    | ε

finally_opt:
    | finally tk_dos_puntos bloque
    | ε

# ===================================================================
# BLOQUE
# ===================================================================

bloque:
    | NEWLINE TAB sentencias TABend

# ===================================================================
# EXPRESIONES (sin recursión izquierda)
# ===================================================================

expresion:
    | expr_or

# OR lógico
expr_or:
    | expr_and expr_or_prime

expr_or_prime:
    | or expr_and expr_or_prime
    | ε

# AND lógico
expr_and:
    | expr_not expr_and_prime

expr_and_prime:
    | and expr_not expr_and_prime
    | ε

# NOT lógico
expr_not:
    | not expr_not
    | expr_comparacion

# Comparaciones
expr_comparacion:
    | expr_aritmetica expr_comp_prime

expr_comp_prime:
    | op_comparacion expr_aritmetica expr_comp_prime
    | ε

op_comparacion:
    | tk_igual
    | tk_distinto
    | tk_menor
    | tk_menor_igual
    | tk_mayor
    | tk_mayor_igual
    | in
    | is not
    | is

# Aritmética: suma y resta
expr_aritmetica:
    | termino expr_arit_prime

expr_arit_prime:
    | tk_suma termino expr_arit_prime
    | tk_resta termino expr_arit_prime
    | ε

# Aritmética: multiplicación, división, módulo
termino:
    | factor termino_prime

termino_prime:
    | tk_mult factor termino_prime
    | tk_div factor termino_prime
    | tk_modulo factor termino_prime
    | ε

# Factor: unarios
factor:
    | tk_suma factor
    | tk_resta factor
    | potencia

# Potencia
potencia:
    | primario potencia_prime

potencia_prime:
    | tk_potencia factor
    | ε

# Primarios
primario:
    | id primario_suffix
    | numero
    | tk_cadena
    | True
    | False
    | None
    | self primario_suffix
    | tk_par_izq expresion tk_par_der
    | lista
    | tupla
    | diccionario
    | conjunto

primario_suffix:
    | tk_par_izq args_opt tk_par_der primario_suffix
    | tk_corchete_izq expresion tk_corchete_der primario_suffix
    | ε

numero:
    | tk_entero numero_decimal

numero_decimal:
    | tk_punto tk_entero
    | ε

# ===================================================================
# ARGUMENTOS Y LISTAS
# ===================================================================

args_opt:
    | args
    | ε

args:
    | expresion args_tail

args_tail:
    | tk_coma expresion args_tail
    | ε

# ===================================================================
# ESTRUCTURAS DE DATOS
# ===================================================================

# Listas
lista:
    | tk_corchete_izq elementos_lista tk_corchete_der

elementos_lista:
    | expresion elementos_lista_tail
    | ε

elementos_lista_tail:
    | tk_coma expresion elementos_lista_tail
    | ε

# Tuplas
tupla:
    | tk_par_izq elementos_tupla tk_par_der

elementos_tupla:
    | expresion tk_coma elementos_tupla_tail
    | ε

elementos_tupla_tail:
    | expresion elementos_tupla_tail_cont
    | ε

elementos_tupla_tail_cont:
    | tk_coma expresion elementos_tupla_tail_cont
    | ε

# Diccionarios
diccionario:
    | tk_llave_izq elementos_dict tk_llave_der

elementos_dict:
    | par_dict elementos_dict_tail
    | ε

par_dict:
    | expresion tk_dos_puntos expresion

elementos_dict_tail:
    | tk_coma par_dict elementos_dict_tail
    | ε

# Conjuntos
conjunto:
    | tk_llave_izq expresion elementos_conjunto_tail tk_llave_der

elementos_conjunto_tail:
    | tk_coma expresion elementos_conjunto_tail
    | ε
